\section*{1. ALGORITHM FOR DESIGN CODE}

\subsection*{1.1 Subsystem Architecture and Reset Initialization}
\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item \textbf{Asynchronous Active-Low Reset:} Upon assertion of \texttt{PRESETn == 0}, synchronously reset all internal state variables on the rising edge of \texttt{PCLK}.
    \item \textbf{Default Configuration Initialization:}
    \begin{itemize}
        \setlength{\itemsep}{0pt}
        \item Set Control Register (\texttt{ADDR\_CRC\_CTRL = 0x00}) to \texttt{0x00000001} (Auto-reset enabled, 32-bit streaming word mode).
        \item Set Polynomial Register (\texttt{ADDR\_CRC\_POLY = 0x08}) to \texttt{0x04C11DB7} (Standard IEEE 802.3 polynomial).
        \item Set Seed/Initial Register (\texttt{ADDR\_CRC\_INIT = 0x0C}) to \texttt{0xFFFFFFFF}.
        \item Set Status Register (\texttt{ADDR\_CRC\_STATUS = 0x04}) to \texttt{0x00000002} (\texttt{READY=1}, \texttt{BUSY=0}, \texttt{ERR=0}).
        \item Clear Interrupt Enable and Status Registers (\texttt{0x18}, \texttt{0x1C}) to \texttt{0x00000000}.
    \end{itemize}
\end{enumerate}

\subsection*{1.2 APB Slave Protocol State Machine and Address Decoding}
\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item \textbf{IDLE State:} Sample \texttt{PSEL} on \texttt{posedge PCLK}. If \texttt{PSEL == 1} and \texttt{PENABLE == 0}, transition to \texttt{SETUP} state.
    \item \textbf{SETUP State:} Latch transfer address \texttt{PADDR[7:0]}, direction flag \texttt{PWRITE}, write data \texttt{PWDATA[31:0]}, and byte strobes \texttt{PSTRB[3:0]}. Transition unconditionally to \texttt{ACCESS} state on next clock.
    \item \textbf{ACCESS State:} Assert \texttt{PENABLE == 1}.
    \begin{itemize}
        \setlength{\itemsep}{0pt}
        \item If programmable wait states $W > 0$, deassert \texttt{PREADY = 0} and decrement internal wait counter until zero.
        \item When wait counter is zero, assert \texttt{PREADY = 1}.
        \item If \texttt{PADDR} exceeds \texttt{ADDR\_MAX (0x1C)} or accesses unmapped offsets, assert \texttt{PSLVERR = 1} and log protocol fault.
        \item On completion of transfer, return to \texttt{IDLE} state if \texttt{PSEL == 0}.
    \end{itemize}
\end{enumerate}

\subsection*{1.3 Configurable 32-bit Parallel Cascaded CRC Engine}
\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item \textbf{Input Bit Reflection (\texttt{refin}):} For each active byte of incoming word \texttt{data\_in[31:0]}, if \texttt{refin == 1}, reverse the bit order:
    \begin{equation*}
        b_k[i] = \text{data\_in}[8k + (7 - i)], \quad \forall i \in [0, 7], \, k \in [0, 3]
    \end{equation*}
    \item \textbf{Parallel Galois Field LFSR Step Cascade:} Process the 32-bit word in four 8-bit combinational stages without CPU wait-states:
    \begin{align*}
        c_1 &= \text{CRC8\_STEP}(c_{\text{accum}}, \, b_0, \, \text{POLY}), \quad
        c_2 = \text{CRC8\_STEP}(c_1, \, b_1, \, \text{POLY}) \\
        c_3 &= \text{CRC8\_STEP}(c_2, \, b_2, \, \text{POLY}), \quad
        c_4 = \text{CRC8\_STEP}(c_3, \, b_3, \, \text{POLY})
    \end{align*}
    where each single-byte step evaluates:
    \begin{equation*}
        c' = \bigoplus_{j=0}^{7} \left( (c \ll 1) \oplus \left( \text{MSB}(c) \ ? \ \text{POLY} : 0 \right) \right)
    \end{equation*}
    \item \textbf{Output Bit Reflection (\texttt{refout}):} If \texttt{refout == 1}, reverse the 32-bit result:
    $r[j] = c_4[31 - j], \, \forall j \in [0, 31]$.
    \item \textbf{Final XOR Inversion (\texttt{xorout}):} If \texttt{xorout\_en == 1}, invert bits: $\text{RESULT} = r \oplus \text{\texttt{0xFFFFFFFF}}$.
    \item \textbf{Status and Handshake Update:} Latch checksum into \texttt{RESULT\_REG}, assert \texttt{STATUS.READY = 1}, deassert \texttt{STATUS.BUSY = 0}, and fire \texttt{IRQ} pulse.
\end{enumerate}

\subsection*{2.1 Architectural Description}
The hardware accelerator subsystem comprises three modular architectural layers:
\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item \textbf{AMBA APB3 Protocol Slave Interface:} Decodes 32-bit aligned byte addresses, handles standard APB \texttt{SETUP} and \texttt{ACCESS} phases, inserts programmable wait states via \texttt{PREADY}, and generates protocol error terminations on invalid register offsets via \texttt{PSLVERR}.
    \item \textbf{Memory-Mapped Control Register File:} Manages eight 32-bit memory-mapped registers supporting run-time polynomial programming, seed value configuration, byte strobe control, and interrupt dispatch.
    \item \textbf{Parallel Cascaded CRC-32 Calculation Core:} Features a 4-stage cascaded LFSR Galois Field network calculating 32-bit CRC values in a single clock cycle ($<3.8\text{ ns}$ delay), along with parameterized bit/byte reflection and XOR-out masking.
\end{enumerate}

\section*{2. ALGORITHM FOR TESTBENCH CODE}

\subsection*{2.1 Clock, Reset, and Virtual Interface Binding}
\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item \textbf{Clock Generation:} Instantiate a continuous clock generator producing a 100 MHz clock signal (\texttt{PCLK}, 10 ns period, 50\% duty cycle).
    \item \textbf{Synchronous Reset Sequence:} Assert active-low reset \texttt{PRESETn = 0} at $t=0\text{ ns}$, maintain for 3 clock periods (30 ns), and release synchronously on a falling edge of \texttt{PCLK}.
    \item \textbf{Virtual Interface Hookup:} Connect physical APB interface signals to the Device Under Test (DUT \texttt{apb\_crc32\_top}) and distribute virtual interface handles to the verification components.
\end{enumerate}

\subsection*{2.2 Master Bus Functional Model (BFM) Driver Algorithm}
\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item \textbf{Idle Phase:} Drive \texttt{PSEL = 0}, \texttt{PENABLE = 0}, \texttt{PADDR = 0}, and wait for stimulus task invocation.
    \item \textbf{APB Write Procedure (\texttt{apb\_write(addr, data, strb)}):}
    \begin{itemize}
        \setlength{\itemsep}{0pt}
        \item At \texttt{posedge PCLK}: Drive \texttt{PSEL = 1}, \texttt{PENABLE = 0}, \texttt{PWRITE = 1}, \texttt{PADDR = addr}, \texttt{PWDATA = data}, \texttt{PSTRB = strb} (\textbf{Setup Phase}).
        \item At next \texttt{posedge PCLK}: Assert \texttt{PENABLE = 1} (\textbf{Access Phase}).
        \item Poll \texttt{PREADY} on each cycle. While \texttt{PREADY == 0}, hold bus signals stable.
        \item Once \texttt{PREADY == 1}: Sample \texttt{PSLVERR}, then deassert \texttt{PSEL = 0} and \texttt{PENABLE = 0}.
    \end{itemize}
    \item \textbf{APB Read Procedure (\texttt{apb\_read(addr, rdata)}):}
    \begin{itemize}
        \setlength{\itemsep}{0pt}
        \item Drive \texttt{PSEL = 1}, \texttt{PENABLE = 0}, \texttt{PWRITE = 0}, \texttt{PADDR = addr}.
        \item At next cycle, assert \texttt{PENABLE = 1}. Wait until \texttt{PREADY == 1}.
        \item Capture \texttt{rdata = PRDATA}, verify \texttt{PSLVERR}, and deassert \texttt{PSEL} and \texttt{PENABLE}.
    \end{itemize}
\end{enumerate}

\subsection*{2.3 Passive Bus Monitor and Analysis Port Dispatch}
\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item Sample bus lines at every rising edge of \texttt{PCLK}.
    \item Detect valid APB transfer completion when \texttt{PSEL == 1 \&\& PENABLE == 1 \&\& PREADY == 1}.
    \item Capture \texttt{PADDR}, \texttt{PWRITE}, \texttt{PWDATA}, \texttt{PRDATA}, and \texttt{PSLVERR}.
    \item Package observed data into transaction packets and broadcast to Scoreboard and Coverage collectors.
\end{enumerate}

\subsection*{2.4 Self-Checking Golden Reference Scoreboard Algorithm}
\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item Maintain shadow state registers mirroring DUT configuration (\texttt{POLY}, \texttt{INIT}, \texttt{REFIN}, \texttt{REFOUT}, \texttt{XOROUT}).
    \item On write to \texttt{ADDR\_CRC\_DATA\_IN (0x10)}, pass input word to algorithmic golden reference model:
    \begin{equation*}
        \text{expected\_crc} = \text{CRC32\_REFERENCE}(\text{data}, \, \text{poly}, \, \text{seed}, \, \text{refin}, \, \text{refout}, \, \text{xorout})
    \end{equation*}
    \item On read from \texttt{ADDR\_CRC\_RESULT (0x14)}, compare DUT output \texttt{PRDATA} with \texttt{expected\_crc}.
    \item If $\text{PRDATA} == \text{expected\_crc}$: Increment \texttt{match\_count}.
    \item If mismatch: Flag test failure, print mismatch report with actual and expected hex values, and increment \texttt{mismatch\_count}.
\end{enumerate}

\subsection*{2.5 Functional Coverage Collection Algorithm}
\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item Define covergroups spanning all 8 address registers, protocol error addresses, read/write directions, byte strobes, and injected wait states ($0 \le W \le 7$).
    \item Sample covergroups on each transfer. Ensure 100\% cross-coverage prior to issuing \texttt{\$finish}.
\end{enumerate}

\subsection*{3.1 Detailed Cycle-by-Cycle Bus Timing Analysis}
The waveform above demonstrates the full execution trajectory of the APB-compliant CRC-32 hardware accelerator operating at 100 MHz:
\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item \textbf{Reset Phase ($0\text{ to }30\text{ ns}$):} \texttt{PRESETn} is held low. All registers reset to defaults: \texttt{PREADY} is high, \texttt{PRDATA} is zero, and \texttt{STATUS\_REG} reads \texttt{0x00000002} (Ready).
    \item \textbf{Register Setup Writes ($30\text{ to }120\text{ ns}$):} The master configures \texttt{ADDR\_CRC\_CTRL (0x00)}, \texttt{ADDR\_CRC\_POLY (0x08)}, and \texttt{ADDR\_CRC\_INIT (0x0C)}. Each write completes in two clock cycles (Setup $\rightarrow$ Access) with \texttt{PREADY=1}.
    \item \textbf{Single-Cycle CRC-32 Calculation ($120\text{ to }210\text{ ns}$):} Input word \texttt{0x34333231} ("1234") is written to \texttt{ADDR\_CRC\_DATA\_IN (0x10)}. The hardware engine evaluates all four 8-bit LFSR stages combinationally within the same cycle ($<3.8\text{ ns}$ propagation delay) and updates the internal accumulator on the next rising clock edge.
    \item \textbf{Readback of Golden Result ($210\text{ to }250\text{ ns}$):} A read transfer to \texttt{ADDR\_CRC\_RESULT (0x14)} asserts \texttt{PSEL=1, PENABLE=1}, returning the exact expected IEEE 802.3 checksum \texttt{0xCBF43926} on \texttt{PRDATA} with zero wait states.
    \item \textbf{Wait-State Insertion ($280\text{ to }360\text{ ns}$):} Control register wait-state bits are programmed to $W=2$. During subsequent transfers, \texttt{PREADY} is held low for 2 extra cycles, verifying full compliance with APB3 wait-state handshaking.
    \item \textbf{Protocol Error Handling ($450\text{ to }480\text{ ns}$):} An unmapped address (\texttt{0x24}) is accessed. The APB slave immediately asserts \texttt{PSLVERR = 1} with \texttt{PREADY = 1}, confirming reliable bus error detection without system lockup.
\end{enumerate}

\section*{3. DESIGN CODE (PART II: PARALLEL CRC ENGINE)}

\begin{lstlisting}[language=Verilog,caption={32-bit Parallel CRC-32 Engine (rtl/crc32\_engine.sv)}]
`timescale 1ns / 1ps
import apb_crc32_pkg::*;

module crc32_engine (
  input  logic        clk, rst_n,
  input  logic [31:0] poly_i, init_val_i,
  input  logic        refin_i, refout_i, xorout_en_i, reset_accum_i, data_valid_i,
  input  logic [3:0]  byte_en_i,
  input  logic [31:0] data_in_i,
  output logic [31:0] crc_result_o,
  output logic        busy_o, done_o
);
  logic [31:0] crc_accum;

  function [31:0] crc8_step(input [31:0] cur, input [7:0] b, input [31:0] poly);
    reg [31:0] c; integer i;
    begin
      c = cur ^ {b, 24'h0};
      for (i = 0; i < 8; i = i + 1) c = c[31] ? ({c[30:0], 1'b0} ^ poly) : {c[30:0], 1'b0};
      crc8_step = c;
    end
  endfunction

  function [7:0]  reflect_byte(input [7:0]  in_b);
    for (int i=0; i<8; i++) reflect_byte[i] = in_b[7-i];
  endfunction

  function [31:0] reflect_word(input [31:0] in_w);
    for (int j=0; j<32; j++) reflect_word[j] = in_w[31-j];
  endfunction

  logic [31:0] s0, s1, s2, s3;
  logic [7:0]  b0, b1, b2, b3;

  always_comb begin
    b0 = refin_i ? reflect_byte(data_in_i[7:0])   : data_in_i[7:0];
    b1 = refin_i ? reflect_byte(data_in_i[15:8])  : data_in_i[15:8];
    b2 = refin_i ? reflect_byte(data_in_i[23:16]) : data_in_i[23:16];
    b3 = refin_i ? reflect_byte(data_in_i[31:24]) : data_in_i[31:24];

    s0 = byte_en_i[0] ? crc8_step(crc_accum, b0, poly_i) : crc_accum;
    s1 = byte_en_i[1] ? crc8_step(s0,        b1, poly_i) : s0;
    s2 = byte_en_i[2] ? crc8_step(s1,        b2, poly_i) : s1;
    s3 = byte_en_i[3] ? crc8_step(s2,        b3, poly_i) : s2;
  end

  always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      crc_accum <= DEFAULT_INIT; busy_o <= 1'b0; done_o <= 1'b0;
    end else begin
      done_o <= 1'b0;
      if (reset_accum_i) begin
        crc_accum <= init_val_i; busy_o <= 1'b0;
      end else if (data_valid_i) begin
        crc_accum <= s3; busy_o <= 1'b0; done_o <= 1'b1;
      end
    end
  end

  logic [31:0] reflected_crc;
  assign reflected_crc = refout_i ? reflect_word(crc_accum) : crc_accum;
  assign crc_result_o  = reflected_crc ^ (xorout_en_i ? DEFAULT_XOROUT : 32'h0);
endmodule
\end{lstlisting}

\begin{enumerate}
    \setlength{\itemsep}{0.5pt}
    \setlength{\parskip}{0pt}
    \item \textbf{APB FSM State Conformance:} The slave FSM strictly follows the AMBA APB v2.0 specification:
    \begin{itemize}
        \setlength{\itemsep}{0pt}
        \item \texttt{IDLE} $\rightarrow$ \texttt{SETUP} occurs when \texttt{PSEL=1} and \texttt{PENABLE=0}.
        \item \texttt{SETUP} $\rightarrow$ \texttt{ACCESS} occurs unconditionally on the next clock edge with \texttt{PENABLE=1}.
        \item In \texttt{ACCESS}, the slave holds \texttt{PREADY=0} for multi-cycle transactions and asserts \texttt{PREADY=1} to conclude the transfer.
    \end{itemize}
    \item \textbf{Programmable Wait-State Verification:} When wait-states are configured ($W=2$), \texttt{PREADY} remains deasserted for exactly 2 additional cycles. The master maintains address, control, and data stable throughout the entire wait interval.
    \item \textbf{Bus Error (\texttt{PSLVERR}) Integrity:} When an unmapped address is driven, \texttt{PSLVERR} asserts concurrently with \texttt{PREADY=1}, terminating the bus transfer gracefully and latching error bit \texttt{STATUS[2]=1} without hanging the bus.
\end{enumerate}

import apb_crc32_pkg::*;

interface apb_interface (input logic pclk, input logic presetn);
  logic [7:0]  paddr = 8'h00;

    output paddr, psel, penable, pwrite, pwdata, pstrb;
    input  prdata, pready, pslverr, irq;
  endclocking

  modport master_mp (clocking master_cb, input pclk, presetn);
  modport slave_mp  (input paddr, psel, penable, pwrite, pwdata, pstrb, pclk, presetn,
                     output prdata, pready, pslverr, irq);
endinterface

module tb_top;
  logic pclk = 0, presetn = 0;
  always #5ns pclk = ~pclk; // 100 MHz clock

  apb_interface vif (pclk, presetn);

  apb_crc32_top u_dut (
    .pclk(vif.pclk), .presetn(vif.presetn), .paddr(vif.paddr),
    .psel(vif.psel), .penable(vif.penable), .pwrite(vif.pwrite),
    .pwdata(vif.pwdata), .pstrb(vif.pstrb), .prdata(vif.prdata),
    .pready(vif.pready), .pslverr(vif.pslverr), .irq(vif.irq)
  );

  int total_trans = 0, match_count = 0, mismatch_count = 0;

  task apb_write(input bit [7:0] addr, input bit [31:0] data, input bit [3:0] strb);
    @(posedge pclk);
    vif.paddr <= addr; vif.pwdata <= data; vif.pstrb <= strb;
    vif.pwrite <= 1'b1; vif.psel <= 1'b1; vif.penable <= 1'b0;
    @(posedge pclk);
    vif.penable <= 1'b1;
    while (!vif.pready) @(posedge pclk);
    @(posedge pclk);
    vif.psel <= 1'b0; vif.penable <= 1'b0;
    total_trans++;
  endtask

  task apb_read(input bit [7:0] addr, output bit [31:0] rdata);
    @(posedge pclk);
    vif.paddr <= addr; vif.pwrite <= 1'b0; vif.psel <= 1'b1; vif.penable <= 1'b0;
    @(posedge pclk);
    vif.penable <= 1'b1;
    while (!vif.pready) @(posedge pclk);
    rdata = vif.prdata;
    @(posedge pclk);
    vif.psel <= 1'b0; vif.penable <= 1'b0;
    total_trans++;
  endtask
endmodule
